#include <iostream>

using namespace std;

string strToLower(string S)
{
    int L = S.length();
    for (int i = 0; i < L; i ++)
    {

        int code = int (S[i]);
        if((code >= -128)&&(code <= -113))
        {
            S[i] = code + 32;
        }
        else if((code >= -112)&&(code <= -97))
        {
            S[i] = code + 80;
        }
        else if (code == 16)
        {
            S[i] = - 15;
        }
    } return S;
}

{
    cout << strToLower("À à Á á Â â Ã ã Ä ä Å å ¨ ¸ Æ æ Ç ç È è É é Ê ê Ë ë Ì ì Í í Î î Ï ï Ð ð Ñ ñ Ò ò Ó ó Ô ô Õ õ Ö ö × ÷ Ø ø Ù ù Ú ú Û û Ü ü Ý ý Þ þ ß ÿ") << endl;


    return 0;
}
