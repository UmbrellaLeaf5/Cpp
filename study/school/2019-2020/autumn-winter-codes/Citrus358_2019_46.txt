string strToLower(string S)
{
    int L = S.length();
    for (int i = 0; i < L; i ++)
    {

        int code = int (S[i]);
        if((code >= -128)&&(code <= -113))
        {
            S[i] = code + 32;
        }
        else if((code >= -112)&&(code <= -97))
        {
            S[i] = code + 80;
        }
        else if (code == 16)
        {
            S[i] = - 15;
        }
    } return S;
}
